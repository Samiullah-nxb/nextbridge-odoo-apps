# Python packages NextContract needs on top of a stock Odoo 19 server.
# Odoo.sh installs this file automatically; on your own server run:
#   pip install -r requirements.txt
# On the official odoo:19.0 Docker image, Debian owns some dependencies
# (typing_extensions, idna), so install with:
#   pip3 install --break-system-packages --ignore-installed -r requirements.txt
# Packages Odoo already ships (lxml, Pillow, reportlab, PyPDF2, openpyxl,
# requests, python-dateutil) are deliberately not listed.

# Required (nextcontract): DOCX templates and drafting, Excel export
python-docx>=1.1.0
docxtpl>=0.16
xlsxwriter

# Required (nextcontract, nextcontract_ai): PDF previews and page rendering
pypdfium2>=4.30.0

# AI (nextcontract_ai, nextcontract_migration). pydantic is needed by every
# real provider; google-genai only by Google Gemini. OpenAI, Anthropic and
# OpenAI-compatible servers use requests, which Odoo ships. AI stays off until
# an administrator configures a provider.
pydantic>=2.0.0
google-genai>=1.0.0

# Keep the version the odoo:19.0 image ships. google-genai would otherwise pull
# a newer cryptography that the image's pyOpenSSL 23.2 cannot load, and Odoo
# then fails to start ("module 'lib' has no attribute 'GEN_EMAIL'").
cryptography==41.0.7
